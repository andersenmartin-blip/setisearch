def identity(s):
    return {'device':s.st_dev,'inode':s.st_ino,'mode':stat.S_IMODE(s.st_mode),
            'file_type_mode':stat.S_IFMT(s.st_mode),'links':s.st_nlink,'bytes':s.st_size,
            'mtime_ns':s.st_mtime_ns,'ctime_ns':s.st_ctime_ns,'allocated_bytes':s.st_blocks*512}

def pins(raw):
    return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest(),
            'git_blob':hashlib.sha1(b'blob '+str(len(raw)).encode()+b'\0'+raw).hexdigest()}

def safe_relative(path):
    if not isinstance(path,str) or not path or path.startswith('/') or '\\' in path or '\0' in path:
        raise AssertionError('unsafe path')
    if any(p in ('','.','..') for p in path.split('/')): raise AssertionError('unsafe component')
    return path

def read_held(path):
    path=Path(path)
    if not path.is_absolute() or os.path.normpath(str(path)) != str(path): raise AssertionError('canonical absolute path')
    guards=[]; parent=os.open('/',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
    guards.append((parent,None,None,os.fstat(parent)))
    fd=None
    try:
        for name in path.parts[1:-1]:
            a=os.stat(name,dir_fd=parent,follow_symlinks=False)
            child=os.open(name,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC,dir_fd=parent)
            b=os.fstat(child)
            if (a.st_dev,a.st_ino,a.st_mode)!=(b.st_dev,b.st_ino,b.st_mode): raise AssertionError('parent substitution')
            guards.append((child,parent,name,b));parent=child
        name=path.name;a=os.stat(name,dir_fd=parent,follow_symlinks=False)
        if not stat.S_ISREG(a.st_mode) or a.st_nlink!=1 or a.st_size>134217728: raise AssertionError('sole regular bounded input')
        fd=os.open(name,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC,dir_fd=parent)
        if identity(a)!=identity(os.fstat(fd)): raise AssertionError('file substitution')
        counts['opened_files']+=1
        raw=bytearray();remaining=a.st_size
        while remaining:
            n=min(65536,remaining)
            if counts['requested_read_bytes']+n>CAP: raise AssertionError('administrative read cap')
            counts['requested_read_bytes']+=n;counts['read_operations']+=1
            chunk=os.read(fd,n);counts['returned_read_bytes']+=len(chunk)
            if not chunk: raise AssertionError('truncated input')
            raw.extend(chunk);remaining-=len(chunk)
        counts['requested_read_bytes']+=1;counts['read_operations']+=1
        suffix=os.read(fd,1);counts['returned_read_bytes']+=len(suffix)
        if suffix or identity(os.fstat(fd))!=identity(a) or identity(os.stat(name,dir_fd=parent,follow_symlinks=False))!=identity(a):
            raise AssertionError('held/named input drift')
        for held,previous,component,original in guards:
            now=os.fstat(held)
            if (now.st_dev,now.st_ino,now.st_mode)!=(original.st_dev,original.st_ino,original.st_mode): raise AssertionError('held directory drift')
            if previous is not None:
                named=os.stat(component,dir_fd=previous,follow_symlinks=False)
                if (named.st_dev,named.st_ino,named.st_mode)!=(now.st_dev,now.st_ino,now.st_mode): raise AssertionError('named directory drift')
        return bytes(raw),identity(a)
    finally:
        if fd is not None:os.close(fd)
        for held,_,_,_ in reversed(guards):os.close(held)

def inventory():
    rows=[];totals={'files':0,'directories':0,'file_bytes':0,'logical_bytes':0,'allocated_bytes':0}
    def walk(path,rel):
        s=os.lstat(path);old=identity(s)
        if not stat.S_ISDIR(s.st_mode) or s.st_dev!=27:raise AssertionError('directory type/device')
        totals['directories']+=1;totals['logical_bytes']+=s.st_size;totals['allocated_bytes']+=s.st_blocks*512
        rows.append({'path':rel,'kind':'directory','identity':old})
        names=sorted(os.listdir(path))
        for name in names:
            safe_relative(name);child=path/name;sub=name if rel=='.' else rel+'/'+name
            a=os.lstat(child)
            if stat.S_ISDIR(a.st_mode):walk(child,sub)
            elif stat.S_ISREG(a.st_mode) and a.st_nlink==1:
                raw,held=read_held(child)
                if held!=identity(a):raise AssertionError('walk opened substituted file')
                row={'path':sub,'kind':'file','identity':held,**pins(raw)}
                rows.append(row);totals['files']+=1;totals['file_bytes']+=len(raw)
                totals['logical_bytes']+=len(raw);totals['allocated_bytes']+=a.st_blocks*512
            else:raise AssertionError('source symlink/hardlink/special object')
        if sorted(os.listdir(path))!=names or identity(os.lstat(path))!=old:raise AssertionError('source directory drift')
    walk(ROOT,'.')
    return {'entries':rows,'totals':totals}
